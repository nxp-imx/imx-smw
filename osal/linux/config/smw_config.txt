/* Example of operation/subsystem configuration file configuring    */
/* the ELE and TEE subsystems.                                      */
/* This configuration defines the subsystem ELE as the subsystem to */
/* use when calling the PSA APIs.                                   */
VERSION=1;
PSA_DEFAULT=ELE:ALT;

/* Defining the HASH operation of the TEE subsystem on top of the    */
/* file is using the cryptographic hash operation with TEE subsystem */
[SECURE_SUBSYSTEM]
    TEE;
    [SECURITY_OPERATION]
        HASH;
    [SECURITY_OPERATION]
        HASH_MULTI_PART;

/* Define the ELE subsystem operations to be executed */
[SECURE_SUBSYSTEM]
    ELE;
    /* No Start/Stop method specified. Default is 1. */
    [SECURITY_OPERATION]
        GENERATE_KEY;
    [SECURITY_OPERATION]
        DERIVE_KEY;
    [SECURITY_OPERATION]
        IMPORT_KEY;
    [SECURITY_OPERATION]
        EXPORT_KEY;
    [SECURITY_OPERATION]
        DELETE_KEY;
    [SECURITY_OPERATION]
        HASH;
    [SECURITY_OPERATION]
        HASH_MULTI_PART;
    [SECURITY_OPERATION]
        SIGN;
        SIGN_ALGO_VALUES=ECDSA;
    [SECURITY_OPERATION]
        VERIFY;
        SIGN_ALGO_VALUES=ECDSA;
    [SECURITY_OPERATION]
        RNG;
    [SECURITY_OPERATION]
        CIPHER;
        KEY_TYPE_VALUES=AES;
        MODE_VALUES=CBC:CFB:CTR:ECB;
    [SECURITY_OPERATION]
        MAC;
    [SECURITY_OPERATION]
        DEVICE_ATTESTATION;
    [SECURITY_OPERATION]
        STORAGE_STORE;
        KEY_TYPE_VALUES=AES;
        MODE_VALUES=CBC:CFB:CTR:ECB;
        MAC_ALGO_VALUES=CMAC;
    [SECURITY_OPERATION]
        STORAGE_RETRIEVE;
    [SECURITY_OPERATION]
        STORAGE_DELETE;
    [SECURITY_OPERATION]
        DEVICE_LIFECYCLE;
    [SECURITY_OPERATION]
        AEAD;
        KEY_TYPE_VALUES=AES;
        MODE_VALUES=CCM:GCM:CHACHA20_POLY1305;
    [SECURITY_OPERATION]
        DEVICE_REPROVISION;

/* Define the TEE subsystem operations to be executed */
[SECURE_SUBSYSTEM]
    TEE;
    /* No Start/Stop method specified. Default is 1. */
    [SECURITY_OPERATION]
        GENERATE_KEY;
    [SECURITY_OPERATION]
        DERIVE_KEY;
    [SECURITY_OPERATION]
        IMPORT_KEY;
    [SECURITY_OPERATION]
        EXPORT_KEY;
    [SECURITY_OPERATION]
        DELETE_KEY;
    [SECURITY_OPERATION]
        SIGN;
        SIGN_ALGO_VALUES=ECDSA:RSA;
        SIGN_TYPE_VALUES=PKCS1_1_5:PSS;
    [SECURITY_OPERATION]
        VERIFY;
        SIGN_ALGO_VALUES=ECDSA:RSA;
        SIGN_TYPE_VALUES=PKCS1_1_5:PSS;
    [SECURITY_OPERATION]
        RNG;
    [SECURITY_OPERATION]
        CIPHER;
        KEY_TYPE_VALUES=AES:DES:DES3:SM4;
        MODE_VALUES=CBC:CTR:CTS:ECB:XTS;
    [SECURITY_OPERATION]
        CIPHER_MULTI_PART;
        KEY_TYPE_VALUES=AES:DES:DES3:SM4;
        MODE_VALUES=CBC:CTR:CTS:ECB:XTS;
    [SECURITY_OPERATION]
        MAC;
    [SECURITY_OPERATION]
        STORAGE_STORE;
    [SECURITY_OPERATION]
        STORAGE_RETRIEVE;
    [SECURITY_OPERATION]
        STORAGE_DELETE;
    [SECURITY_OPERATION]
        AEAD;
        KEY_TYPE_VALUES=AES;
        MODE_VALUES=CCM:GCM;
    [SECURITY_OPERATION]
        AEAD_MULTI_PART;
        KEY_TYPE_VALUES=AES;
        MODE_VALUES=CCM:GCM;
    [SECURITY_OPERATION]
        ASYMM_ENCRYPT;
        ENC_ALGO_VALUES=RSA;
        MODE_VALUES=PKCS1_1_5:OAEP:NO_PAD;
    [SECURITY_OPERATION]
        ASYMM_DECRYPT;
        ENC_ALGO_VALUES=RSA;
        MODE_VALUES=PKCS1_1_5:OAEP:NO_PAD;
